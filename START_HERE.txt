AVAILABLE COMMANDS:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

docker-compose run --rm shellcoding list
  → Show available encoders

docker-compose run --rm shellcoding encode --technique xor --input payload.hex
  → Encode with specific technique

docker-compose run --rm shellcoding compile --technique xor --input encoded.c
  → Compile encoded shellcode

docker-compose run --rm shellcoding pipeline --technique xor --input payload.hex
  → Full pipeline: encode + compile + test

docker-compose run --rm shellcoding oscp
  → Show OSCP quick reference

docker-compose run --rm shellcoding --help
  → Show all available commands
